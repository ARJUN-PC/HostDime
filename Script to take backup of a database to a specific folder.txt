#!/bin/bash

# Define database credentials
DB_NAME="ramadbkc_RamaDBK"
BACKUP_DIR="/home/ramadbkcom/public_html/dbbkp"

# Get the current date and time to append to the backup file name
DATE=$(date +"%Y%m%d%H%M%S")

# Create backup file name
BACKUP_FILE="${BACKUP_DIR}/${DB_NAME}_${DATE}.sql"

# Print debugging information
echo "Running backup for database: ${DB_NAME}"
echo "Backup directory: ${BACKUP_DIR}"
echo "Backup file: ${BACKUP_FILE}"

# Ensure the backup directory exists
if [ ! -d "${BACKUP_DIR}" ]; then
  echo "Backup directory does not exist. Creating: ${BACKUP_DIR}"
  mkdir -p ${BACKUP_DIR}
fi

# Run mysqldump to take the backup and capture any error output
mysqldump ${DB_NAME} > ${BACKUP_FILE} 2>&1

# Check if the backup was successful
if [ $? -eq 0 ]; then
  echo "Backup successful: ${BACKUP_FILE}"
  
  # Check if the backup file was created
  if [ -f "${BACKUP_FILE}" ]; then
    echo "Backup file created: ${BACKUP_FILE}"
    
    # Remove existing backups (except the latest one) after creating a new one
    echo "Removing existing backups..."
    find ${BACKUP_DIR} -name "${DB_NAME}_*.sql" ! -name "$(basename ${BACKUP_FILE})" -exec rm -f {} \;

  else
    echo "Backup file was not created as expected."
  fi
else
  echo "Backup failed! Output of mysqldump command:"
  cat ${BACKUP_FILE}
  # Optionally, remove the failed backup file
  rm -f ${BACKUP_FILE}
fi
