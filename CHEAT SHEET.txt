#!/bin/bash

# Set variables for database connection
PGUSER=postgres
PGDATABASE=ems

# Set the path where you want to store the backup files
BACKUP_DIR=/backup/PG_Backup/

# Get current date and time
datestamp=$(date +'%Y-%m-%d')
timestamp=$(date +'%H%M')

# Execute pg_dump command to dump the database
pg_dump -U "$PGUSER" -d "$PGDATABASE" > "PG_Database_backup-EMS$BACKUP_DIR/$PGDATABASE"_"$datestamp"_"$timestamp".sql
